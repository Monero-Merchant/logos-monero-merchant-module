cmake_minimum_required(VERSION 3.22)
project(merchant_delivery LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID)
    message(FATAL_ERROR "The merchant JNI transport targets Android. Basecamp uses delivery_module through the SDK.")
endif()
set(DELIVERY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/artifacts/delivery-${ANDROID_ABI}" CACHE PATH "Pinned liblogosdelivery package")
if(NOT EXISTS "${DELIVERY_ROOT}/include/liblogosdelivery.h")
    message(FATAL_ERROR "Delivery artifacts missing. Run pos/scripts/prepare-native.sh android first.")
endif()
file(GLOB CBOR_SOURCES "${DELIVERY_ROOT}/include/tinycbor/*.c")
add_library(tinycbor STATIC ${CBOR_SOURCES})
target_include_directories(tinycbor PUBLIC "${DELIVERY_ROOT}/include" "${DELIVERY_ROOT}/include/tinycbor")
add_library(logosdelivery SHARED IMPORTED)
set_target_properties(logosdelivery PROPERTIES IMPORTED_LOCATION "${DELIVERY_ROOT}/lib/liblogosdelivery.so")
add_library(merchant_delivery SHARED merchant_delivery.cpp)
target_include_directories(merchant_delivery PRIVATE "${DELIVERY_ROOT}/include")
target_link_libraries(merchant_delivery PRIVATE tinycbor logosdelivery)
target_compile_definitions(merchant_delivery PRIVATE $<$<CONFIG:Debug>:MERCHANT_DELIVERY_DEBUG>)
target_link_libraries(merchant_delivery PRIVATE $<$<CONFIG:Debug>:log>)
target_link_options(merchant_delivery PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
